2. Node.js + Express Server (`server/index.js`) 

const express = require('express');
const cors = require('cors');
const mongoose = require('mongoose');
require('dotenv').config();

const app = express();
app.use(cors());
app.use(express.json());

// MongoDB Connection
mongoose.connect(process.env.MONGO_URI, {
    useNewUrlParser: true,
    useUnifiedTopology: true
}).then(() => console.log('MongoDB ga muvaffaqiyatli ulandi'))
  .catch(err => console.error('DB Xatolik:', err));

// Routes
app.use('/api/quiz', require('./routes/quizRoutes'));
app.use('/api/mentor', require('./routes/mentorRoutes'));

const PORT = process.env.PORT || 5000;
app.listen(PORT, () => console.log(`Server ${PORT}-portda ishlamoqda`));

3. MongoDB Question Model (`server/models/Question.js`) 

const mongoose = require('mongoose');

const QuestionSchema = new mongoose.Schema({
    topic: { type: String, required: true }, // masalan: "HTML Form"
    difficulty: { type: String, enum: ['easy', 'medium', 'hard'], default: 'medium' },
    question: { type: String, required: true },
    codeSnippet: { type: String, default: "" },
    options: [{ type: String, required: true }],
    correctIndex: { type: Number, required: true },
    explanation: { type: String, required: true }
}, { timestamps: true });

module.exports = mongoose.model('Question', QuestionSchema);

4. Vercel va Render Hostingga Joylash (Deploy Guide)

  1. GitHub Repositori Yaratish: Barcha kodlarni GitHub repositoriyasiga yuklang (`git push`).
  2. MongoDB Atlas: mongodb.com saytidan bepul MongoDB klaster yaratib connection string URI oling.
  3. Backendni Deploy qilish (Render.com / Railway): Render.com ga kirib `server` papkasini tanlang, `MONGO_URI` va `JWT_SECRET` environment variable larini kiriting.
  4. Frontendni Vercel ga joylash: Vercel.com sahifasiga o'tib repositoriyani tanlang va bir bosishda Publish qiling!
